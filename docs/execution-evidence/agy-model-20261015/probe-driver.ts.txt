const source = new Context()
let stopped: string | null = null
let sequence = 0, tail = Promise.resolve(), traceWrites = Promise.resolve()
const trace = (event: object) => {
  traceWrites = traceWrites.then(() => appendFile(join(root, 'provider-calls.jsonl'), JSON.stringify(event) + '\n'))
}
class SerialInstalledAgyAdapter extends LlmAdapter {
  override providerInfo(provider: string) { return source.llm.listProviders().find(row => row.id === provider)! }
  override listModels(provider: string) { return source.llm.listModels(provider) }
  override resolveModel(provider: string, id: string, signal?: AbortSignal) { return source.llm.resolveModelInfo(provider, id, signal) }
  override providerRetryPolicy() { return resolveRetryPolicy({ mode: 'normal', maxRetries: 0 }, 'experiment') }
  override async *stream(options: GenerateOptions): AsyncIterable<StreamChunk> {
    assert.equal(options.provider, 'agy'); assert.equal(options.model, model.id)
    assert.equal(options.reasoningEffort, 'medium')
    const previous = tail
    let release!: () => void
    tail = new Promise<void>(done => { release = done })
    await previous
    try {
      options.signal?.throwIfAborted()
      if (stopped) throw new Error('AGY experiment stopped before queued provider dispatch')
      const call = ++sequence
      const prepared = await source.llm.prepareCall({ provider: 'agy', model: model.id,
        reasoningEffort: ReasoningEffortId('medium'), maxTokens: options.maxTokens }, options.signal)
      assert.equal(prepared.config.provider, 'agy'); assert.equal(prepared.config.model, model.id)
      assert.equal(prepared.config.reasoningEffort, 'medium')
      trace({ kind: 'dispatch', call, at: Date.now(), provider: 'agy', model: model.id,
        reasoningEffort: prepared.config.reasoningEffort, maxTokens: prepared.config.maxTokens })
      for await (const chunk of prepared.stream(options)) {
        if (chunk.type === 'finish') {
          const failure = chunk.reason.kind === 'error' ? chunk.reason.failure : null
          if (failure && ([401, 403, 429].includes(failure.status ?? 0) || /AUTH|RATE_LIMIT|QUOTA|PERMISSION/.test(failure.code))) {
            stopped = 'provider-auth-access-or-quota-rejected'
          }
          trace({ kind: 'finish', call, at: Date.now(), reason: chunk.reason.kind,
            ...(failure ? { code: failure.code, status: failure.status ?? null } : {}) })
        }
        yield chunk
      }
    } finally { release() }
  }
}

const completed: Array<Awaited<ReturnType<typeof runShiftingEvidence>> & { directory: string }> = []
try {
  await source.plugin(LlmRuntime)
  await source.plugin(plugin)
  assert(source.llm.listProviders().some(row => row.id === 'agy'))
  const info = await source.llm.resolveModelInfo('agy', model.id)
  assert(info.reasoning?.efforts.some(row => row.id === 'medium'))
  for (const seed of seeds) {
    assert(isDeepStrictEqual(sourceHashes, await shiftingSourceHashes()), 'Source drift before seed')
    const directory = join(root, `${model.id}-adaptive-seed-${seed}`)
    const report = await runShiftingEvidence({ ...conditions, seed, model, mode: 'adaptive', directory, purpose: 'adaptive-probe' },
      undefined, { provenance: { provider: 'agy', plugin: 'dsh-agy', profile: 'installed-desktop-plugin',
        modelImport: 'installed-dsh-agy', mount: 'dsh-installed-plugin', model: model.id, reasoningEffort: 'medium' },
      async liveProviderMount(ctx) { ctx.llm.registerAdapter(['agy'], new SerialInstalledAgyAdapter()) } })
    completed.push({ ...report, directory })
    if (report.failures.some(row => [401,403,429].includes(row.status ?? 0) || /AUTH|RATE_LIMIT|QUOTA|PERMISSION/.test(row.code))) stopped = 'provider-auth-access-or-quota-rejected'
    if (report.cleanup !== 'released') stopped = 'cleanup-not-released'
    await traceWrites
    await writeFile(join(root, 'batch.json'), JSON.stringify({ ...plan, completed, stopped, actualProviderDispatches: sequence }, null, 2) + '\n')
    console.log('AGY_PROBE_RUN ' + JSON.stringify({ seed, passed: report.passed, phase1: report.phase1Correct, phase2: report.phase2Correct,
      calls: report.issuedModelCalls, providerDispatches: sequence, stop: report.stopReason, failures: report.failures,
      cleanup: report.cleanup, completed: completed.length, planned: seeds.length, stopped }))
    if (stopped) break
  }
} finally { await source.fiber.dispose(); await traceWrites }
console.log('AGY_PROBE_DONE ' + JSON.stringify({ completed: completed.length, planned: seeds.length, stopped, actualProviderDispatches: sequence }))
if (stopped) process.exitCode = 1
